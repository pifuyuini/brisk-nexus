% 基本模版的封面
\ExplSyntaxOn

% 定义命令：将全角逗号替换为 \quad
\NewDocumentCommand{\YsyCoverAuthors}{m}
{
	\ysyauthors_format:n { #1 }
}

\cs_new:Npn \ysyauthors_format:n #1
{
	\tl_set:Nn \l_tmpa_tl { #1 }
	\tl_replace_all:Nnn \l_tmpa_tl { ， } { \quad }
	\tl_use:N \l_tmpa_tl
}

\ExplSyntaxOff


% 封面页设计
\newcommand{\YsyCoverPage}{%
	\begin{titlepage}
		\newgeometry{margin=0in}
		\pagestyle{empty}
		\parindent=0pt
		
		% 插图
		\begin{figure}
			\centering
			\includegraphics[width=\linewidth]{\CoverIllustration}
		\end{figure}
		
		\begin{flushright}
			% 标题
			{\huge\bfseries \TitleName} \hspace*{6em} \\[0.2cm]
			\rule{0.618\textwidth}{5pt} \hspace*{6em} \\[0.4cm]
			
			% 作者
			\LARGE\emph{\YsyCoverAuthors{\AuthorName}} \hspace*{4.1em} \\[0.5cm]
			
			% 地址
			\large\textbf{\Address} \hspace*{5em}
		\end{flushright}
		
		% Bottom of the page
		\begin{center}
			\vfill
			
			\Large\texttt{\textcolor{StructureColor!80!black}{\Preface}}
			
			\hspace*{\fill}
			
			{\LARGE \today}
			
			\hspace*{\fill}
		\end{center}
		
		\restoregeometry
	\end{titlepage}
}
