% A（必要支持包）
\RequirePackage{amsmath} % 提供增强的数学公式排版功能，如 align、gather 等环境。
\RequirePackage{amssymb} % 包含更多数学符号
\RequirePackage{geometry} % 页面调整
\RequirePackage[no-math]{fontspec} % 字体（主要是英文）
\RequirePackage{fancyhdr} % 自定义页眉页脚样式，可添加章节标题、页码等。
\RequirePackage{graphicx} % 插入图片的核心包
\RequirePackage{xcolor} % 颜色
\RequirePackage[skip=2pt]{caption} % 自定义图片/表格标题格式。
\RequirePackage{enumerate} % 增强 enumerate 环境，允许指定编号格式。
\RequirePackage{enumitem} % 列表设置依赖包
\RequirePackage{multirow} % 在表格中合并多行单元格。
\RequirePackage{float} % 用于控制浮动体（如表格、图片）的放置，提供 H 选项让浮动体固定在当前位置。
\RequirePackage{tcolorbox} % 颜色盒子
\RequirePackage{hyperref} % 超链接
\RequirePackage{cleveref} % 引用
\RequirePackage{titlesec} % 调整章节标题
\RequirePackage{listings} % 代码环境
\RequirePackage{expl3} % 字符串替换

% B（用户便利包）
\RequirePackage{adjustbox} % 调整表格大小
\RequirePackage{physics} % 提供数学物理相关命令，如 \bra{}、\ket{}。
\RequirePackage{appendix} % 创建附录部分，允许 \appendix 轻松管理附录编号。
\usepackage{authblk} % 管理多作者信息
\usepackage{subfig} % 创建多个子图
\RequirePackage{tikz} % 矢量图绘制
\RequirePackage{tabularx} % 创建自动调整列宽的表格，比 tabular 更灵活。
