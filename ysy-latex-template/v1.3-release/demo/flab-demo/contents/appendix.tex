\subsection{公式表}
\[
\Gamma(z)\Gamma(1 - z) = \frac{\pi}{\sin(\pi z)}
\]

\[
\Gamma(2z) = 2^{2z - 1} \sqrt{\pi} \, \Gamma(z) \Gamma\left(z + \frac{1}{2}\right)
\]

\[
x! \approx \sqrt{2\pi x} \left( \frac{x}{e} \right)^x \quad \text{(Re}\, x \gg 1)
\]

\[
J_\nu(x) = \sum_{k=0}^{\infty} \frac{(-1)^k}{k! \Gamma(k + \nu + 1)} \left( \frac{x}{2} \right)^{2k + \nu}
\]

\[
N_\nu(x) = \lim_{\alpha \to \nu} \frac{J_\alpha(x) \cos(\alpha\pi) - J_{-\alpha}(x)}{\sin(\alpha \pi)}
\]

\[
\frac{d}{dx} \left[ x^\nu Z_\nu(x) \right] = x^\nu Z_{\nu - 1}(x)
\]

\[
\frac{d}{dx} \left[ x^{-\nu} Z_\nu(x) \right] = -x^{-\nu} Z_{\nu + 1}(x)
\]

\[
Z_{\nu - 1}(x) - Z_{\nu + 1}(x) = 2 Z'_\nu(x)
\]

\[
Z_{\nu - 1}(x) + Z_{\nu + 1}(x) = \frac{2\nu}{x} Z_\nu(x)
\]

\[
J_\nu(x) \approx \sqrt{\frac{2}{\pi x}} \cos\left( x - \frac{\nu\pi}{2} - \frac{\pi}{4} \right)
\quad (x \gg \sqrt{|\nu^2 - \frac{1}{4}|})
\]

\[
N_\nu(x) \approx \sqrt{\frac{2}{\pi x}} \sin\left( x - \frac{\nu\pi}{2} - \frac{\pi}{4} \right)
\quad (x \gg \sqrt{|\nu^2 - \frac{1}{4}|})
\]

\[
J_m(x) = \frac{1}{2\pi} \int_a^{a+2\pi} e^{i(x \sin \theta - m \theta)}\, d\theta
\quad (m \in \mathbb{Z},\; a \in \mathbb{R})
\]

\[
e^{\frac{x}{2} \left(t - \frac{1}{t}\right)} = \sum_{n=-\infty}^{\infty} J_n(x) t^n
\]

\[
\int_0^1 x J_m(\mu_i x) J_m(\mu_j x) dx = 
\frac{\delta_{ij}}{2} [J_{m+1}(\mu_i)]^2
\quad (\mu_i: J_m \text{的第}i\text{个正零点})
\]

\[
\int_0^1 x J_m(\lambda_i x) J_m(\lambda_j x) dx = 
\frac{\delta_{ij}}{2} \left(1 - \frac{m^2}{\lambda_i^2}\right) [J_m(\lambda_i)]^2
\quad (\lambda_i: J_m' \text{的第}i\text{个正零点})
\]

\[
\int_0^\infty J_m(k_1 r) J_m(k_2 r) r\,dr = \frac{\delta(k_1 - k_2)}{k_1}
\quad (k_1, k_2 > 0)
\]

\[
j_\ell(x) = \sqrt{\frac{\pi}{2x}} J_{\ell + \frac{1}{2}}(x)
\]

\[
n_\ell(x) = \sqrt{\frac{\pi}{2x}} N_{\ell + \frac{1}{2}}(x)
\]

\[
j_\ell(x) = (-x)^\ell \left( \frac{1}{x} \frac{d}{dx} \right)^\ell \left( \frac{\sin x}{x} \right)
\]

\[
n_\ell(x) = -(-x)^\ell \left( \frac{1}{x} \frac{d}{dx} \right)^\ell \left( \frac{\cos x}{x} \right)
\]

\[
Y_{\ell m}(\theta, \phi) = 
\frac{1}{2^\ell \ell!}
\sqrt{\frac{(2\ell + 1)}{4\pi} \cdot \frac{(\ell - m)!}{(\ell + m)!}}
\left[ \sin^m \theta \left( \frac{1}{\sin \theta} \frac{d}{d\theta} \right)^{\ell + m} \sin^{2\ell} \theta \right]
e^{im\phi}
\]

\[
P_\ell(\cos \theta) = \sqrt{\frac{4\pi}{2\ell + 1}} Y_{\ell 0}(\theta, \phi)
\]

\[
P_\ell(x) = \sum_{k=0}^\ell \frac{(\ell + k)!}{(k!)^2 (\ell - k)!} \left( \frac{x - 1}{2} \right)^k
\]

\[
P_\ell(x) = \sum_{k=0}^\ell \frac{(-1)^{\ell - k} (\ell + k)!}{(k!)^2 (\ell - k)!} \left( \frac{x + 1}{2} \right)^k
\]

\[
P_{2n}(x) = \frac{1}{2^{2n}} \sum_{k=0}^n \frac{(-1)^{n - k} [2(n + k)]!}{(n + k)! (n - k)! (2k)!} x^{2k}
\]

\[
P_{2n+1}(x) = \frac{1}{2^{2n+1}} \sum_{k=0}^n \frac{(-1)^{n - k} [2(n + k + 1)]!}{(n + k + 1)! (n - k)! (2k + 1)!} x^{2k + 1}
\]

\[
P_\ell(x) = \frac{1}{2^\ell \ell!} \frac{d^\ell}{dx^\ell} (x^2 - 1)^\ell
\]

\[
(2\ell + 1) x P_\ell(x) = (\ell + 1) P_{\ell + 1}(x) + \ell P_{\ell - 1}(x)
\]

\[
(2\ell + 1) P_\ell(x) = P'_{\ell + 1}(x) - P'_{\ell - 1}(x)
\]

\[
\int_{-1}^1 P_m(x) P_n(x)\, dx = \frac{2}{2m + 1} \delta_{mn}
\]

\[
\frac{1}{\sqrt{1 + t^2 - 2xt}} = \sum_{\ell = 0}^\infty P_\ell(x) t^\ell
\quad \text{for } t < \left| x \pm \sqrt{x^2 - 1} \right|
\]

\[
P_\ell(\mathbf{n}_1 \cdot \mathbf{n}_2) = \frac{4\pi}{2\ell + 1} \sum_{m = -\ell}^\ell Y_{\ell m}^*(\mathbf{n}_1) Y_{\ell m}(\mathbf{n}_2)
